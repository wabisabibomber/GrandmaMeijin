cmake_minimum_required(VERSION 3.24)
project(GrandmaMeijin VERSION 1.1.0 LANGUAGES CXX RC)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 / MSVC is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# /MTでC++ランタイムを静的リンクし、再頒布パッケージを不要にする。
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /utf-8 /permissive- /EHsc)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_library(meijin_core ClickController.cpp Settings.cpp)
target_link_libraries(meijin_core PUBLIC user32 shell32 ole32 windowsapp)
target_include_directories(meijin_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set(ICON_RESOURCE "")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Assets/app.ico")
  set(ICON_RESOURCE "101 ICON DISCARDABLE \"${CMAKE_CURRENT_SOURCE_DIR}/Assets/app.ico\"")
endif()
configure_file(App.rc.in App.rc @ONLY)
add_executable(GrandmaMeijin WIN32 Main.cpp Dialogs.rc "${CMAKE_CURRENT_BINARY_DIR}/App.rc")
target_link_libraries(GrandmaMeijin PRIVATE meijin_core comctl32)
target_link_options(GrandmaMeijin PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
set_target_properties(GrandmaMeijin PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/artifacts/bin")
include(CTest)
if(BUILD_TESTING)
  add_executable(NativeChecks tests/Checks.cpp)
  target_link_libraries(NativeChecks PRIVATE meijin_core)
  add_test(NAME NativeChecks COMMAND NativeChecks)
  add_executable(NativeUiChecks WIN32 Main.cpp Dialogs.rc "${CMAKE_CURRENT_BINARY_DIR}/App.rc")
  target_compile_definitions(NativeUiChecks PRIVATE MEIJIN_UI_CHECK)
  target_link_libraries(NativeUiChecks PRIVATE meijin_core comctl32 gdi32)
  target_link_options(NativeUiChecks PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
  add_test(NAME NativeUiChecks COMMAND NativeUiChecks)
endif()
install(TARGETS GrandmaMeijin RUNTIME DESTINATION .)
